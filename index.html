Integrasi Google Sheets Terpusat

Ikuti langkah berikut agar Data Presensi, Mapel, Guru, Admin, dan Lokasi otomatis tersimpan ke multi-sheet (Sheet 1, 2, 3, dst) dan tersinkronisasi di semua HP:

  1. Buat Google Sheet baru di Google Drive Anda (misal beri nama: Presensi & Master Data SMKN 1 Negara Batin).
  2. Di Google Sheet tersebut, klik menu Ekstensi / Extensions > Apps Script.
  3. Hapus seluruh kode yang ada di Apps Script, lalu Salin & Tempel kode Script di bawah ini:
function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var result = {
      result: "success",
      subjects: parseSubjects(getSheetValues(ss, "Data Mapel")),
      teachers: parseTeachers(getSheetValues(ss, "Data Guru")),
      admins: parseAdmins(getSheetValues(ss, "Data Admin")),
      locations: parseLocations(getSheetValues(ss, "Lokasi & QR"))
    };
    return ContentService.createTextOutput(JSON.stringify(result))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ result: "error", error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var contents = JSON.parse(e.postData.contents);
    var action = contents.action;
    var data = contents.data;

    if (action === "ADD_ATTENDANCE") {
      var sheet = getOrCreateSheet(ss, "Presensi Guru", [
        "ID Transaksi", "Tanggal", "Jam Scan", "NIP", "Nama Guru", 
        "Peran", "Mata Pelajaran", "Materi Pembelajaran", "Jam Ke", 
        "Waktu KBM", "Lokasi", "Jarak (m)", "Status", "Keterangan Terlambat"
      ]);
      sheet.appendRow([
        data.id, data.date, data.time, "'" + data.userNip, data.userName,
        data.roleType, data.mapel, data.materi || "-",
        "Jam " + data.jamDari + " s/d " + data.jamSampai,
        data.timeStart + " - " + data.timeEnd,
        data.locationName, data.distance, data.status, data.lateNote || ""
      ]);
    } else if (action === "SYNC_SUBJECTS") {
      updateSheetRows(ss, "Data Mapel", ["No", "Nama Mata Pelajaran"], data.map(function(item, i) {
        return [i + 1, item];
      }));
    } else if (action === "SYNC_TEACHERS") {
      updateSheetRows(ss, "Data Guru", ["ID", "NIP", "Nama Guru", "Mata Pelajaran", "Username", "Password", "Status"], data.map(function(t) {
        return [t.id, "'" + t.nip, t.name, t.mapel, t.user, t.pass, t.status || "ACTIVE"];
      }));
    } else if (action === "SYNC_ADMINS") {
      updateSheetRows(ss, "Data Admin", ["ID", "Nama Administrator", "Username", "Password", "Status"], data.map(function(a) {
        return [a.id, a.name, a.user, a.pass, a.status || "ACTIVE"];
      }));
    } else if (action === "SYNC_LOCATIONS") {
      updateSheetRows(ss, "Lokasi & QR", ["ID", "Kode QR", "Nama Lokasi", "Latitude", "Longitude", "Radius (m)"], data.map(function(l) {
        return [l.id, l.code, l.name, l.lat, l.lng, l.radius];
      }));
    }

    return ContentService.createTextOutput(JSON.stringify({ result: "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ result: "error", error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function getOrCreateSheet(ss, name, headers) {
  var sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
    if (headers) {
      sheet.appendRow(headers);
      sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold").setBackground("#0f2942").setFontColor("#ffffff");
    }
  }
  return sheet;
}

function updateSheetRows(ss, sheetName, headers, rows) {
  var sheet = getOrCreateSheet(ss, sheetName, headers);
  sheet.clearContents();
  sheet.appendRow(headers);
  sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold").setBackground("#0f2942").setFontColor("#ffffff");
  if (rows && rows.length > 0) {
    sheet.getRange(2, 1, rows.length, headers.length).setValues(rows);
  }
}

function getSheetValues(ss, sheetName) {
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) return [];
  var vals = sheet.getDataRange().getValues();
  return vals.length > 1 ? vals : [];
}

function parseSubjects(vals) {
  var res = [];
  for (var i = 1; i < vals.length; i++) {
    if (vals[i][1]) res.push(String(vals[i][1]));
  }
  return res;
}

function parseTeachers(vals) {
  var res = [];
  for (var i = 1; i < vals.length; i++) {
    if (vals[i][0]) {
      res.push({
        id: String(vals[i][0]),
        nip: String(vals[i][1]).replace(/^'/, ''),
        name: String(vals[i][2]),
        mapel: String(vals[i][3]),
        user: String(vals[i][4]),
        pass: String(vals[i][5]),
        status: String(vals[i][6] || 'ACTIVE')
      });
    }
  }
  return res;
}

function parseAdmins(vals) {
  var res = [];
  for (var i = 1; i < vals.length; i++) {
    if (vals[i][0]) {
      res.push({
        id: String(vals[i][0]),
        name: String(vals[i][1]),
        user: String(vals[i][2]),
        pass: String(vals[i][3]),
        status: String(vals[i][4] || 'ACTIVE')
      });
    }
  }
  return res;
}

function parseLocations(vals) {
  var res = [];
  for (var i = 1; i < vals.length; i++) {
    if (vals[i][0]) {
      res.push({
        id: String(vals[i][0]),
        code: String(vals[i][1]),
        name: String(vals[i][2]),
        lat: parseFloat(vals[i][3]) || -4.8213,
        lng: parseFloat(vals[i][4]) || 104.9123,
        radius: parseInt(vals[i][5]) || 50,
        active: true
      });
    }
  }
  return res;
}
